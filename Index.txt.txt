    <script>
        /* ==========================================
           CONSTANTES Y LÍMITES FÍSICOS
           ========================================== */
        const ABSOLUTE_ZERO = {
            C: -273.15,
            F: -459.67,
            K: 0
        };

        const UNIT_NAMES = {
            C: 'grados Celsius',
            F: 'grados Fahrenheit',
            K: 'Kelvin'
        };

        const UNIT_SYMBOLS = {
            C: '°C',
            F: '°F',
            K: 'K'
        };

        /* Elementos del DOM */
        const tempInput = document.getElementById('tempInput');
        const fromUnitSelect = document.getElementById('fromUnit');
        const toUnitSelect = document.getElementById('toUnit');
        const convertBtn = document.getElementById('convertBtn');
        const resetBtn = document.getElementById('resetBtn');
        const swapBtn = document.getElementById('swapBtn');
        const errorMessage = document.getElementById('errorMessage');
        const resultSection = document.getElementById('resultSection');
        textResult = document.getElementById('resultText');
        const stateBadge = document.getElementById('stateBadge');
        const formulaDisplay = document.getElementById('formulaDisplay');
        const explanationText = document.getElementById('explanationText');

        /* ==========================================
           EVENT LISTENERS
           ========================================== */
        convertBtn.addEventListener('click', processConversion);
        resetBtn.addEventListener('click', resetApp);
        swapBtn.addEventListener('click', swapUnits);

        // Permite realizar la conversión al presionar Enter
        tempInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter') processConversion();
        });

        // Conversión automática opcional al cambiar parámetros o escribir
        tempInput.addEventListener('input', autoConvert);
        fromUnitSelect.addEventListener('change', autoConvert);
        toUnitSelect.addEventListener('change', autoConvert);

        function autoConvert() {
            if (tempInput.value.trim() !== '') {
                processConversion();
            } else {
                hideError();
                resultSection.style.display = 'none';
            }
        }

        /* ==========================================
           LÓGICA PRINCIPAL DE CONVERSIÓN
           ========================================== */
        function processConversion() {
            const rawValue = tempInput.value.trim();
            const fromUnit = fromUnitSelect.value;
            const toUnit = toUnitSelect.value;

            // Validación de campo vacío o no numérico
            if (rawValue === '' || isNaN(rawValue)) {
                showError('Por favor, ingresa un número válido.');
                return;
            }

            const val = parseFloat(rawValue);

            // Validación del cero absoluto
            if (val < ABSOLUTE_ZERO[fromUnit]) {
                showError(`La temperatura no puede ser inferior al Cero Absoluto (${ABSOLUTE_ZERO[fromUnit]} ${UNIT_SYMBOLS[fromUnit]}).`);
                return;
            }

            hideError();

            // Cálculo dinámico
            const { result, formula, explanation, celsiusValue } = convertTemperature(val, fromUnit, toUnit);

            // Renderizado de resultados
            displayResults(val, fromUnit, result, toUnit, formula, explanation, celsiusValue);
        }

        function convertTemperature(value, from, to) {
            let celsius;
            let result;
            let formula = '';
            let explanation = '';

            // 1. Convertir unidad de origen a Celsius
            if (from === 'C') {
                celsius = value;
            } else if (from === 'F') {
                celsius = (value - 32) * 5 / 9;
            } else if (from === 'K') {
                celsius = value - 273.15;
            }

            // 2. Convertir Celsius a unidad de destino
            if (from === to) {
                result = value;
                formula = `${UNIT_SYMBOLS[from]} = ${UNIT_SYMBOLS[to]}`;
                explanation = 'La unidad de origen y destino es la misma, por lo que el valor no cambia.';
            } else if (from === 'C' && to === 'F') {
                result = (value * 9 / 5) + 32;
                formula = '°F = (°C × 9/5) + 32';
                explanation = `Multiplicamos ${value} por 9/5 (o 1.8) y luego sumamos 32.`;
            } else if (from === 'C' && to === 'K') {
                result = value + 273.15;
                formula = 'K = °C + 273.15';
                explanation = `Sumamos 273.15 a los ${value} °C para obtener la escala absoluta Kelvin.`;
            } else if (from === 'F' && to === 'C') {
                result = (value - 32) * 5 / 9;
                formula = '°C = (°F - 32) × 5/9';
                explanation = `Restamos 32 a ${value} °F y luego multiplicamos por 5/9 (o dividimos entre 1.8).`;
            } else if (from === 'F' && to === 'K') {
                result = ((value - 32) * 5 / 9) + 273.15;
                formula = 'K = (°F - 32) × 5/9 + 273.15';
                explanation = `Convertimos ${value} °F primero a Celsius y luego sumamos 273.15 para llegar a Kelvin.`;
            } else if (from === 'K' && to === 'C') {
                result = value - 273.15;
                formula = '°C = K - 273.15';
                explanation = `Restamos 273.15 al valor de ${value} K para obtener grados Celsius.`;
            } else if (from === 'K' && to === 'F') {
                result = ((value - 273.15) * 9 / 5) + 32;
                formula = '°F = (K - 273.15) × 9/5 + 32';
                explanation = `Convertimos ${value} K primero a Celsius restando 273.15, y luego convertimos esa cifra a Fahrenheit.`;
            }

            return {
                result: parseFloat(result.toFixed(2)),
                formula,
                explanation,
                celsiusValue: celsius
            };
        }

        /* ==========================================
           ACTUALIZACIÓN DE INTERFAZ & UI DYNÁMICA
           ========================================== */
        function displayResults(origVal, origUnit, resVal, destUnit, formula, explanation, celsiusVal) {
            textResult.textContent = `${origVal} ${UNIT_SYMBOLS[origUnit]} = ${resVal} ${UNIT_SYMBOLS[destUnit]}`;
            formulaDisplay.textContent = formula;
            explanationText.textContent = explanation;

            // Determinar estado térmico según la temperatura en Celsius
            resultSection.classList.remove('state-cold', 'state-warm', 'state-hot');

            if (celsiusVal < 10) {
                resultSection.classList.add('state-cold');
                stateBadge.textContent = 'Sensación: Fría ❄️';
            } else if (celsiusVal >= 10 && celsiusVal <= 28) {
                resultSection.classList.add('state-warm');
                stateBadge.textContent = 'Sensación: Templada / Agradable 🌿';
            } else {
                resultSection.classList.add('state-hot');
                stateBadge.textContent = 'Sensación: Caliente 🔥';
            }

            resultSection.style.display = 'block';
        }

        function swapUnits() {
            const tempFrom = fromUnitSelect.value;
            fromUnitSelect.value = toUnitSelect.value;
            toUnitSelect.value = tempFrom;
            autoConvert();
        }

        function loadExample(val, from, to) {
            tempInput.value = val;
            fromUnitSelect.value = from;
            toUnitSelect.value = to;
            processConversion();
        }

        function showError(msg) {
            errorMessage.textContent = msg;
            errorMessage.style.display = 'block';
            resultSection.style.display = 'none';
        }

        function hideError() {
            errorMessage.style.display = 'none';
            errorMessage.textContent = '';
        }

        function resetApp() {
            tempInput.value = '';
            fromUnitSelect.value = 'C';
            toUnitSelect.value = 'F';
            hideError();
            resultSection.style.display = 'none';
            tempInput.focus();
        }
    </script>
